Bash script é um tipo de linguagem de programação do tipo `shell script` que é interpretada por interpretadores de comandos de um sistema Linux-like.

É o tipo mais comum de shell em distribuições baseadas em Linux.
# Funcionalidades básicas:

```bash
#!/bin/bash
# o comentário acima especifica a "shebang" que instrui o sistema operacional qual tipo de interpretador deve ser utilizado para executar o script. Neste caso, o interpretador "bash" encontrado no caminho "/bin/bash"

# variáveis
ip="192.168.0.10"
read other_ip # permite input do usuário (salva em $other_ip)
ping -c1 $ip # exemplo de uso de uma variável

# ---

read number
let sum=$number+10 # tratar `number` como um número e somar 10. Neste caso não pode colocar espaços
sum2=$(( $number + 10 )) # ou coloca entre $(()) que faz a mesma coisa
echo "$sum | $sum2" # "20 | 20"

# ---

# condicional
username="admin"
if [[ $username == "admin" ]]; then
	echo "You're admin"
elif [[ $username == "gabrier" ]]; then
	echo "You're gabrier"
else
	echo "You're some other user"
fi

if (( $number > 10 )); then
	echo "$number is greater than 10"
fi

# verificação de conteúdo null
if [[ -z $target ]] || [[ -z $ext ]]; then
	echo "Usage: $0 <target> <filetype>"
	exit
fi

# ---

# repetição
i=10
while (( $i>0 )); do
	echo "$i"
	i=$(( $i-1 ))
done

for i in {0..10}; do
	echo "$i"
done

for word in $(cat wordlist.txt); do
	echo "$word"
done
```
